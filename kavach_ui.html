import os import sys import asyncio from fastapi import FastAPI from fastapi.middleware.cors import CORSMiddleware from pydantic import BaseModel from google import genai if sys.platform == "win32": asyncio.set_event_loop_policy(asyncio.WindowsSelectorEventLoopPolicy()) app = FastAPI(title="Cyber Kavach AI Engine") app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"], ) # Pulls from Render Environment Variable or falls back to local string GEMINI_API_KEY = os.environ.get("GEMINI_API_KEY", "AIzaSy...") client = genai.Client(api_key=GEMINI_API_KEY) class ThreatRequest(BaseModel): message: str @app.post("/analyze-threat") def analyze_threat(payload: ThreatRequest): prompt = f""" You are the Cyber Kavach Emergency AI Incident Solver. The user reported the following incident: "{payload.message}" Analyze the threat and provide a clear, formatted resolution: 🚨 THREAT ASSESSMENT: [CRITICAL / HIGH RISK / SUSPICIOUS / SAFE] 🔍 CRIME MECHANISM: [How this scam operates in 1-2 concise sentences] 🛡️ STEP-BY-STEP SOLUTION FOR USER: 1. [Immediate emergency step] 2. [Remediation & 1930 / cybercrime.gov.in guidance] 3. [Evidence preservation step] """ # Try flash first, fall back to stable lite if overloaded for model_name in ["gemini-2.5-flash", "gemini-2.5-flash-lite"]: try: response = client.models.generate_content( model=model_name, contents=prompt ) return {"status": "success", "analysis": response.text} except Exception: continue return {"status": "error", "analysis": "AI Service temporarily busy. Please click Solve again in 5 seconds."} if __name__ == "__main__": import uvicorn uvicorn.run(app, host="127.0.0.1", port=8000)
